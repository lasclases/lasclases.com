/* Dise~ne un programa con las siguientes caracteristicas:
 1.- Debe Solicitar al usuario 2 matrices de 3x3
 2.- Debe conterner 3 funciones:
     a.- Una que reciba una matriz de 3x3 y no devuelva nada.
         En la Matriz recibida se colocaran los datos que llenaran la matriz.
         Estos deben ser pedidos desde la misma funcion.
         (Esta funcion hara lo requerido en el punto 1).
     b.- Una funcion que reciba una matriz de 3x3 y la muestre por pantalla.
     c.- Una funcion que reciba 3 matrices, sume las 2 primeras y devuelva el
         resultado en la tercera.
 3.- Las tres matrices seran declaradas local al bloque main().
 4.- Todas las funciones deben ser llamadas desde main().
*/

#include<stdio.h>

void llenaMat(int m[3][3]);
void muestra(int m[3][3]);
void suma(int a[3][3], int b[3][3], int c[3][3]);

int main(void) {
    int a[3][3]={0}, b[3][3]={0}, c[3][3]={0};
    printf("Primero llenemos las matrices\n");
    printf("\nPrimero la Matriz a\n");
    llenaMat(a);
    printf("La Matriz que introdujo es:\n");
    muestra(a);
    printf("\nAhora llenaremos b\n");
    llenaMat(b);
    printf("La Matriz que introdujo es:\n");
    muestra(b);
    printf("Al sumar ambas Matrices obtenemos:\n");
    suma(a,b,c);
    muestra(c);
    return 0;
}

void llenaMat(int m[3][3]){
    int x, y;
    for(x=0;x<=2;x++)
	for(y=0;y<=2;y++){
	    printf("Introduzca el Elemento[%i][%i]: ",x+1,y+1);
	    scanf("%i", &m[x][y] );
	}
}

void muestra(int m[3][3]){
    int x, y;
    for(x=0;x<=2;x++){
	for(y=0;y<=2;y++) printf("%6i",m[x][y]);
        printf("\n");
    }

}

void suma(int a[3][3], int b[3][3], int c[3][3]){
    int x,y;
    for(x=0;x<=2;x++)
	for(y=0;y<=2;y++)
	    c[x][y]=a[x][y]+b[x][y];
}
